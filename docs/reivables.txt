#[async_trait]
impl AccountRepository for PostgresAccountRepo {
    /// # Purpose
    /// Create a new user and insert into the database. User may be employee, admin, or owner.
    /// Purpose of user is authentication and authorization to access and use the application.
    /// User will be used to perform sign in/signup, DBMS CRUD functions and any relevant jobs.
    ///
    /// # Errors
    ///
    /// This function will return an error if user cannot be created.
    async fn create(&self, user_id: sqlx::types::Uuid, acc: &CreateAccount) -> Result<Account, AppError> {
        let account = sqlx::query_as::<_, Account>(
            "INSERT INTO accounts (name, type, user_id) VALUES ($1, $2, $3) RETURNING *"
        )
        .bind(&acc.name)
        .bind(&acc.type_)
        .bind(user_id)
        .fetch_one(&self.pool)
        .await?;
        Ok(account)
    }

    async fn find_by_id(&self, id: sqlx::types::Uuid, user_id: sqlx::types::Uuid) -> Result<Option<Account>, AppError> {
        let account = sqlx::query_as::<_, Account>(
            "SELECT * FROM accounts WHERE id = $1 AND user_id = $2"
        )
        .bind(id)
        .bind(user_id)
        .fetch_optional(&self.pool)
        .await?;
        Ok(account)
    }

    async fn find_by_user(&self, user_id: sqlx::types::Uuid) -> Result<Vec<Account>, AppError> {
        let accounts = sqlx::query_as::<_, Account>("SELECT * FROM accounts WHERE user_id = $1")
            .bind(user_id)
            .fetch_all(&self.pool)
            .await?;
        Ok(accounts)
    }

    async fn update_balance(&self, id: sqlx::types::Uuid, delta: BigDecimal) -> Result<(), AppError> {
        sqlx::query("UPDATE accounts SET balance = balance + $1 WHERE id = $2")
            .bind(delta)
            .bind(id)
            .execute(&self.pool)
            .await?;
        Ok(())
    }

    async fn update_account_info(&self, id: Uuid, user_id: Uuid, updates: &CreateAccount) -> Result<Account, AppError> {
        let account = sqlx::query_as::<_, Account>(
            "UPDATE accounts SET name = $1, type = $2 WHERE id = $3 AND user_id = $4 RETURNING *",
        )
        .bind(&updates.name)
        .bind(&updates.type_)
        .bind(id)
        .bind(user_id)
        .fetch_one(&self.pool)
        .await
        .map_err(|e| {
            if e.as_database_error()
                .and_then(|db| db.constraint())
                .map(|c| c == "accounts_type_check")
                .unwrap_or(false)
            {
                AppError::Validation("Invalid account type".into())
            } else {
                AppError::Database(e)
            }
        })?;

        Ok(account)
    }

    fn pool(&self)->&PgPool{
        &self.pool
    }
}

use crate::errors::AppError;
use crate::features::accounts::repository::AccountRepository;
use crate::models::account::{Account, CreateAccount};
use uuid::Uuid;

pub struct AccountingService<R: AccountRepository> {
    account_repo: R,
}

impl<R: AccountRepository> AccountingService<R> {
    pub fn new(account_repo: R) -> Self {
        Self { account_repo }
    }

    pub async fn create_account(
        &self,
        user_id: Uuid,
        acc: &CreateAccount,
    ) -> Result<Account, AppError> {
        let valid_types = ["asset", "liability", "equity", "revenue", "expense"];
        if !valid_types.contains(&acc.type_.as_str()) {
            return Err(AppError::Validation("Invalid account type".into()));
        }
        self.account_repo.create(user_id, acc).await
    }

    pub async fn get_accounts(&self, user_id: Uuid) -> Result<Vec<Account>, AppError> {
        self.account_repo.find_by_user(user_id).await
    }

    pub async fn get_account(&self, id: Uuid, user_id: Uuid) -> Result<Account, AppError> {
        self.account_repo
            .find_by_id(id, user_id)
            .await?
            .ok_or(AppError::NotFound("Account not found".into()))
    }

    pub async fn update_account(
        &self,
        id: Uuid,
        user_id: Uuid,
        updates: &CreateAccount,
    ) -> Result<Account, AppError> {
        let valid_types = ["asset", "liability", "equity", "revenue", "expense"];
        if !valid_types.contains(&updates.type_.as_str()) {
            return Err(AppError::Validation("Invalid account type".into()));
        }

        // Now delegate to repo
        self.account_repo.update_account_info(id, user_id, updates).await
    }

    pub async fn delete_account(&self, id: Uuid, user_id: Uuid) -> Result<(), AppError> {
        let result = sqlx::query("DELETE FROM accounts WHERE id = $1 AND user_id = $2")
            .bind(id)
            .bind(user_id)
            .execute(self.account_repo.pool())
            .await?;

        if result.rows_affected() == 0 {
            return Err(AppError::NotFound("Account not found".into()));
        }
        Ok(())
    }
}

pub fn create_router(state: Arc<AppState>) -> Router {
    let protected_transactions: Router<Arc<AppState>> =
        crate::features::transactions::handlers::router()
            .layer(middleware::from_fn(auth_middleware));
    let protected_accounts: Router<Arc<AppState>> =
        crate::features::accounts::handlers::router().layer(middleware::from_fn(auth_middleware));

    Router::new()
        // CHECK THAT SERVER IS UP AND RUNING
        .route(
            "/",
            get(|| async { "Yey! Service Up And Runing Successfully!\n" }),
        )
        // PUBLIC ROUTES (no auth required)
        .nest("/api/auth", crate::features::auth::handlers::router())
        // PROTECTED ROUTES (require auth)
        .nest("/api/transactions", protected_transactions)
        .nest("/api/accounts", protected_accounts)
        // CORS & global state
        .layer(Extension(state.clone()))
        .layer(CorsLayer::permissive())
        // Get log info for audit trail
        .layer(
            TraceLayer::new_for_http()
                .make_span_with(|request: &Request| {
                    let user_id = request
                        .extensions()
                        .get::<Authenticated>()
                        .map(|auth: &Authenticated| auth.0.to_string())
                        .unwrap_or_else(|| "anonymous".to_string());

                    tracing::info_span!(
                        "request",
                        method = %request.method(),
                        uri = %request.uri(),
                        user_id = %user_id,
                    )
                })
                .on_response(
                    |response: &Response, latency: Duration, _span: &tracing::Span| {
                        tracing::info!(
                            status = response.status().as_u16(),
                            latency = format!("{:.2?}", latency),
                            "response sent"
                        );
                    },
                )
                .on_failure(
                    |error: ServerErrorsFailureClass, latency: Duration, _span: &tracing::Span| {
                        tracing::error!(
                            status = 500,
                            latency = format!("{:.2?}", latency),
                            error = %error,
                            "request failed"
                        );
                    },
                ),
        )
        .with_state(state)
        .fallback(handler_404) // ← ADD THIS
}

// errors.rs
use axum::{http::StatusCode, response::IntoResponse};
use thiserror::Error;
use serde::Serialize;

#[derive(Serialize)]
struct ErrorResponse {
    status: u16,
    error: String,
    message: String,
}

#[derive(Error, Debug)]
pub enum AppError {
    #[error("Database error: {0}")]
    Database(#[from] sqlx::Error),
    #[error("Auth error: {0}")]
    Auth(String),
    #[error("Validation error: {0}")]
    Validation(String),
    #[error("Not found")]
    NotFound(String),
}

impl IntoResponse for AppError {
    fn into_response(self) -> axum::response::Response {
        let (status, error_type, message): (StatusCode, &str, String) = match self {
            AppError::Database(e) => (
                StatusCode::INTERNAL_SERVER_ERROR,
                "Database error",
                e.to_string(),
            ),
            AppError::Auth(msg) => (
                StatusCode::UNAUTHORIZED,
                "Authentication error",
                msg,
            ),
            AppError::Validation(msg) => (
                StatusCode::BAD_REQUEST,
                "Validation error",
                msg,
            ),
            AppError::NotFound(msg) => (
                StatusCode::NOT_FOUND,
                "Not Found",
                msg,
            ),
        };

        let error_response: ErrorResponse = ErrorResponse {
            status: status.as_u16(),
            error: error_type.to_string(),
            message,
        };

        (status, axum::Json(error_response)).into_response()
    }
}

# 2 — Core accounting tables (double-entry foundation)

Use `bigserial` primary keys and `NUMERIC(18,2)` for money. Adjust precision if you need fractions.

```sql
-- accounts: chart of accounts
CREATE TABLE accounting.accounts (
    id BIGSERIAL PRIMARY KEY,
    code TEXT NOT NULL UNIQUE,    -- e.g. "1.1.01"
    name TEXT NOT NULL,
    type TEXT NOT NULL,           -- Asset, Liability, Equity, Revenue, Expense
    parent_id BIGINT REFERENCES accounting.accounts(id) ON DELETE SET NULL,
    normal_balance TEXT NOT NULL, -- 'DR' or 'CR'
    is_contra BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

CREATE INDEX ON accounting.accounts (type);
CREATE INDEX ON accounting.accounts (code);

-- transactions: header/journal
CREATE TABLE accounting.transactions (
    id BIGSERIAL PRIMARY KEY,
    txn_uuid UUID DEFAULT uuid_generate_v4() NOT NULL UNIQUE,
    txn_date DATE NOT NULL,
    reference TEXT,
    description TEXT,
    created_by BIGINT, -- fk to system.users
    created_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
    posted BOOLEAN DEFAULT TRUE,  -- keep for drafts/approval workflows
    module TEXT                 -- e.g., 'invoice', 'payment', 'journal'
);

CREATE INDEX ON accounting.transactions (txn_date);
CREATE INDEX ON accounting.transactions (module);

-- transaction entries (double-entry lines)
CREATE TABLE accounting.transaction_entries (
    id BIGSERIAL PRIMARY KEY,
    transaction_id BIGINT NOT NULL REFERENCES accounting.transactions(id) ON DELETE CASCADE,
    account_id BIGINT NOT NULL REFERENCES accounting.accounts(id),
    line_no INT NOT NULL,
    amount NUMERIC(18,2) NOT NULL CHECK (amount >= 0),
    debit NUMERIC(18,2) NOT NULL DEFAULT 0 CHECK (debit >= 0),
    credit NUMERIC(18,2) NOT NULL DEFAULT 0 CHECK (credit >= 0),
    memo TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);

CREATE INDEX ON accounting.transaction_entries (account_id);
CREATE INDEX ON accounting.transaction_entries (transaction_id);
CREATE INDEX ON accounting.transaction_entries (created_at);
```

Notes:

* We store `debit` and `credit` explicitly (either one is zero). `amount` can be redundant but useful; you can enforce `amount = debit + credit`.
* `line_no` maintains order for UI readability.


# 3 — Stored procedure: Atomic posting that enforces balance

Instead of depending on application to post many entries correctly, expose a single stored procedure that accepts JSONB lines and atomically inserts header + entries and validates that debits == credits.

```sql
-- function to post a transaction with JSONB entries
CREATE OR REPLACE FUNCTION accounting.post_transaction(
    p_txn_date DATE,
    p_reference TEXT,
    p_description TEXT,
    p_created_by BIGINT,
    p_module TEXT,
    p_lines JSONB  -- array of {account_code or account_id, debit, credit, memo}
) RETURNS BIGINT
LANGUAGE plpgsql
AS $$
DECLARE
    v_txn_id BIGINT;
    v_total_debits NUMERIC(18,2) := 0;
    v_total_credits NUMERIC(18,2) := 0;
    v_line RECORD;
    v_account_id BIGINT;
    v_line_no INT := 0;
BEGIN
    -- compute sums from JSON input
    FOR v_line IN SELECT * FROM jsonb_to_recordset(p_lines)
        AS (account_ref TEXT, debit NUMERIC, credit NUMERIC, memo TEXT)
    LOOP
        v_total_debits := v_total_debits + COALESCE(v_line.debit, 0);
        v_total_credits := v_total_credits + COALESCE(v_line.credit, 0);
    END LOOP;

    IF v_total_debits IS NULL OR v_total_credits IS NULL THEN
        RAISE EXCEPTION 'Debits or credits missing';
    END IF;

    IF v_total_debits <> v_total_credits THEN
        RAISE EXCEPTION 'Unbalanced transaction: debits (%) != credits (%)', v_total_debits, v_total_credits;
    END IF;

    -- insert header
    INSERT INTO accounting.transactions(txn_date, reference, description, created_by, module)
    VALUES (p_txn_date, p_reference, p_description, p_created_by, p_module)
    RETURNING id INTO v_txn_id;

    -- insert lines
    FOR v_line IN SELECT * FROM jsonb_to_recordset(p_lines)
            AS (account_ref TEXT, debit NUMERIC, credit NUMERIC, memo TEXT)
    LOOP
        v_line_no := v_line_no + 1;

        -- resolve account_ref: allow numeric id or code
        BEGIN
            v_account_id := (v_line.account_ref)::BIGINT;
        EXCEPTION WHEN others THEN
            SELECT id INTO v_account_id FROM accounting.accounts WHERE code = v_line.account_ref LIMIT 1;
            IF v_account_id IS NULL THEN
                RAISE EXCEPTION 'Account % not found', v_line.account_ref;
            END IF;
        END;

        INSERT INTO accounting.transaction_entries(
            transaction_id, account_id, line_no, amount, debit, credit, memo
        ) VALUES (
            v_txn_id, v_account_id, v_line_no, COALESCE(v_line.debit,0) + COALESCE(v_line.credit,0), COALESCE(v_line.debit,0), COALESCE(v_line.credit,0), v_line.memo
        );
    END LOOP;

    RETURN v_txn_id;
END;
$$;
```

Usage example (from psql):

```sql
SELECT accounting.post_transaction(
  '2025-11-03'::date,
  'INV-0001',
  'Invoice #INV-0001 sale to ACME',
  1,
  'invoice',
  '[
    {"account_ref":"1.2.1", "debit": 1000, "credit": 0, "memo":"A/R - ACME"},
    {"account_ref":"4.1.1", "debit": 0, "credit": 1000, "memo":"Revenue - Product"}
  ]'::jsonb
);
```

Notes:

* `account_ref` accepts either account `id` (numeric) or `code` (text like `"1.2.1"`).
* The function throws if unbalanced or if an account ref is not found.
* This pattern forces all postings through a single server-side API — consistent, auditable, and safe.

```sql
CREATE INDEX ON receivables.invoice_items (invoice_id);

-- payments
CREATE TABLE receivables.payments (
    id BIGSERIAL PRIMARY KEY,
    invoice_id BIGINT REFERENCES receivables.invoices(id) ON DELETE SET NULL,
    customer_id BIGINT NOT NULL REFERENCES receivables.customers(id) ON DELETE RESTRICT,
    payment_date DATE NOT NULL,
    amount NUMERIC(18,2) NOT NULL,
    method TEXT,  -- Cash, Bank, MobileMoney
    reference TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT now()
);
CREATE INDEX ON receivables.payments (customer_id);
CREATE INDEX ON receivables.payments (payment_date);
```

Procedure example to create invoice and post GL (high level):

```sql
CREATE OR REPLACE FUNCTION receivables.create_invoice_and_post_gl(
    p_invoice_number TEXT,
    p_customer_id BIGINT,
    p_issue_date DATE,
    p_due_date DATE,
    p_total NUMERIC,
    p_items JSONB,  -- optional items array
    p_created_by BIGINT
) RETURNS BIGINT LANGUAGE plpgsql AS $$
DECLARE
    v_invoice_id BIGINT;
    v_lines JSONB;
BEGIN
    INSERT INTO receivables.invoices(invoice_number, customer_id, issue_date, due_date, total_amount, balance_due)
    VALUES (p_invoice_number, p_customer_id, p_issue_date, p_due_date, p_total, p_total)
    RETURNING id INTO v_invoice_id;

    -- Optionally store items
    IF p_items IS NOT NULL THEN
        INSERT INTO receivables.invoice_items (invoice_id, description, quantity, unit_price, total)
        SELECT v_invoice_id, it->>'description', (it->>'quantity')::NUMERIC, (it->>'unit_price')::NUMERIC, (it->>'total')::NUMERIC
        FROM jsonb_array_elements(p_items) AS t(it);
    END IF;

    -- build JSON lines for GL: debit A/R, credit Revenue
    SELECT jsonb_build_array(
        jsonb_build_object('account_ref', '1.2.1', 'debit', p_total, 'credit', 0, 'memo', format('AR - invoice %s', p_invoice_number)),
        jsonb_build_object('account_ref', '4.1.1', 'debit', 0, 'credit', p_total, 'memo', format('Revenue invoice %s', p_invoice_number))
    ) INTO v_lines;

    PERFORM accounting.post_transaction(p_issue_date, p_invoice_number, 'Invoice posted', p_created_by, 'invoice', v_lines);

    -- update customer balance (simple)
    UPDATE receivables.customers SET current_balance = current_balance + p_total WHERE id = p_customer_id;

    RETURN v_invoice_id;
END;
$$;
```

Adjust `account_ref` to your actual account codes for AR and Revenue as seeded.

# 5 — Audit logging (immutable audit trail)

Simple trigger-based audit log for sensitive tables (transactions, transaction_entries, invoices, payments).

```sql
CREATE TABLE system.audit_logs (
    id BIGSERIAL PRIMARY KEY,
    schema_name TEXT,
    table_name TEXT,
    record_id TEXT,
    operation TEXT,
    changed_by BIGINT,
    changed_at TIMESTAMP WITH TIME ZONE DEFAULT now(),
    old_data JSONB,
    new_data JSONB
);

CREATE OR REPLACE FUNCTION system.audit_trigger_fn() RETURNS trigger AS $$
BEGIN
    IF TG_OP = 'INSERT' THEN
        INSERT INTO system.audit_logs(schema_name, table_name, record_id, operation, changed_by, new_data)
        VALUES (TG_TABLE_SCHEMA, TG_TABLE_NAME, NEW.*::text, TG_OP, NULL, to_jsonb(NEW));
        RETURN NEW;
    ELSIF TG_OP = 'UPDATE' THEN
        INSERT INTO system.audit_logs(schema_name, table_name, record_id, operation, changed_by, old_data, new_data)
        VALUES (TG_TABLE_SCHEMA, TG_TABLE_NAME, NEW.*::text, TG_OP, NULL, to_jsonb(OLD), to_jsonb(NEW));
        RETURN NEW;
    ELSIF TG_OP = 'DELETE' THEN
        INSERT INTO system.audit_logs(schema_name, table_name, record_id, operation, changed_by, old_data)
        VALUES (TG_TABLE_SCHEMA, TG_TABLE_NAME, OLD.*::text, TG_OP, NULL, to_jsonb(OLD));
        RETURN OLD;
    END IF;
END;
$$ LANGUAGE plpgsql;

-- attach audit trigger to important tables
CREATE TRIGGER trg_audit_transactions
AFTER INSERT OR UPDATE OR DELETE ON accounting.transactions
FOR EACH ROW EXECUTE FUNCTION system.audit_trigger_fn();

CREATE TRIGGER trg_audit_tx_entries
AFTER INSERT OR UPDATE OR DELETE ON accounting.transaction_entries
FOR EACH ROW EXECUTE FUNCTION system.audit_trigger_fn();

CREATE TRIGGER trg_audit_invoices
AFTER INSERT OR UPDATE OR DELETE ON receivables.invoices
FOR EACH ROW EXECUTE FUNCTION system.audit_trigger_fn();

CREATE TRIGGER trg_audit_payments
AFTER INSERT OR UPDATE OR DELETE ON receivables.payments
FOR EACH ROW EXECUTE FUNCTION system.audit_trigger_fn();
```

Note: record_id uses `NEW.*::text`—you may instead prefer a specific primary key field for clarity.

# 6 — Indexing, partitioning & performance tips

* Index `transaction_entries` on `(account_id, created_at)` for ledger queries.
* Use partial indexes for active accounts: `CREATE INDEX ON accounting.transaction_entries (account_id) WHERE created_at > now() - interval '2 years';` if needed.
* For very high volumes, partition `accounting.transaction_entries` by `transaction_date` or `created_at` (monthly/quarterly). Use native Postgres partitioning.
* VACUUM regularly; run `VACUUM (FULL)` minimally, rely on autovacuum and tune thresholds for heavy write loads.
* Use connection pooling (PgBouncer) for Rust to avoid connection storms.

---

# 7 — Migrations & deployment workflow

* Use a migration tool: `sqitch`, `migra`, `diesel migration`, or `sqlx` migrations. Keep schema SQL in versioned migration scripts.
* Seed minimal chart of accounts with a first migration; more accounts can be imported by CSV later.
* Back up with `pg_dump` schema-by-schema if you want portability.
